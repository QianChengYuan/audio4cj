

/// 表条目数（每条目 8 个十六进制字符）
protected let GBK_TABLE_ENTRIES: Int64 = GBK_TABLE_HEX.size / 8

/// 二分查找 GBK 双字节码位对应的 Unicode 码位；表中无该码位返回 -1
protected func gbkLookup(gbkCode: Int64): Int64 {
    var lo: Int64 = 0
    var hi: Int64 = GBK_TABLE_ENTRIES - 1
    while (lo <= hi) {
        let mid = (lo + hi) / 2
        // 每条目 8 字符：前 4 位是 GBK 码（首字节+次字节），后 4 位是 Unicode
        let cur = hex4(GBK_TABLE_HEX[mid * 8..mid * 8 + 4])
        if (cur == gbkCode) {
            return hex4(GBK_TABLE_HEX[mid * 8 + 4..mid * 8 + 8])
        }
        if (cur < gbkCode) {
            lo = mid + 1
        } else {
            hi = mid - 1
        }
    }
    -1
}

/// 4 位十六进制字符串转数值（表内数据恒为合法 hex；非法输入返回 -1）
protected func hex4(s: String): Int64 {
    let bytes = s.toArray()
    if (Int64(bytes.size) != 4) {
        return -1
    }
    var v: Int64 = 0
    for (b in bytes) {
        let d: Int64 = match (b) {
            case c where c >= 0x30 && c <= 0x39 => Int64(c) - 0x30
            case c where c >= 0x41 && c <= 0x46 => Int64(c) - 0x41 + 10
            case c where c >= 0x61 && c <= 0x66 => Int64(c) - 0x61 + 10
            case _ => return -1
        }
        v = v * 16 + d
    }
    return v
}
